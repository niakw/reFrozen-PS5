cmake_minimum_required(VERSION 3.25)
project(refrozen_ps5_host_gates LANGUAGES CXX)

enable_testing()

option(REFROZEN_PS5_COMPILE_ONLY "Compile the fixture into an object without linking a host executable" OFF)
if(REFROZEN_PS5_COMPILE_ONLY)
  add_library(guest32_ranges_compile OBJECT guest32_ranges_test.cpp)
  set(_fixture guest32_ranges_compile)
else()
  add_executable(guest32_ranges_test guest32_ranges_test.cpp)
  add_test(NAME guest32_ranges COMMAND guest32_ranges_test)
  set(_fixture guest32_ranges_test)
endif()
target_compile_features(${_fixture} PRIVATE cxx_std_20)
target_include_directories(${_fixture} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../ps5/host")
if(MSVC)
  target_compile_options(${_fixture} PRIVATE /W4 /WX)
else()
  target_compile_options(${_fixture} PRIVATE -Wall -Wextra -Werror -Wpedantic -Wconversion)
endif()
